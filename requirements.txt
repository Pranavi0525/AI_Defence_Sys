# requirements.txt

# PRODUCTION runtime dependencies only. This is what actually needs to be
# installed for the API and the pipeline scripts to run.
#
# xgboost is deliberately NOT listed here even though it's required at
# runtime -- its Linux wheel unconditionally depends on nvidia-nccl-cu13
# (a ~290MB GPU/distributed-training package), which this CPU-only
# deployment never uses. It's installed separately with --no-deps by
# scripts/install.sh (see that file for the full explanation). Always run
# `bash scripts/install.sh` instead of `pip install -r requirements.txt`
# directly, or xgboost will be missing.
#
# google-genai and pytest/pytest-cov are intentionally excluded: they are
# not imported anywhere in the API or pipeline code (only in the test
# suite) and don't belong in a production install. See requirements-dev.txt.
